1. Create a thread and display a message



#include <stdio.h>
#include <pthread.h>

void *display(void *arg)
{
    printf("Hello from thread!\n");
    return NULL;
}

int main()
{
    pthread_t t;

    pthread_create(&t, NULL, display, NULL);
    pthread_join(t, NULL);

    printf("Main thread executed.\n");
    return 0;
}

4. Display processes in order of arrival

#include <stdio.h>

int main()
{
    int n, i, p[20], at[20], j, temp;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("Enter process ID and arrival time: ");
        scanf("%d%d", &p[i], &at[i]);
    }

    for(i = 0; i < n - 1; i++)
    {
        for(j = 0; j < n - i - 1; j++)
        {
            if(at[j] > at[j + 1])
            {
                temp = at[j];
                at[j] = at[j + 1];
                at[j + 1] = temp;

                temp = p[j];
                p[j] = p[j + 1];
                p[j + 1] = temp;
            }
        }
    }

    printf("Process execution order:\n");
    for(i = 0; i < n; i++)
        printf("P%d ", p[i]);

    return 0;
}

        printf("Hello from child process!\n");

    {

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *next;
};

int main()
{
    struct Node *head = NULL, *temp, *newNode;
    int n, i, value;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        newNode = malloc(sizeof(struct Node));

        if(newNode == NULL)
            return 1;

        printf("Enter data: ");
        scanf("%d", &value);

        newNode->data = value;
        newNode->next = NULL;

        if(head == NULL)
            head = newNode;
        else
            temp->next = newNode;

        temp = newNode;
    }

    printf("Linked List: ");
    temp = head;

    while(temp != NULL)
    {
        printf("%d -> ", temp->data);
        temp = temp->next;
    }

    printf("NULL");

    while(head != NULL)
    {
        temp = head;
        head = head->next;
        free(temp);
    }

    return 0;
}




9. Perform push and pop operations on a stack


#include <stdio.h>
#define MAX 5

int stack[MAX], top = -1;

void push(int value)
{
    if(top == MAX - 1)
        printf("Stack Overflow\n");
    else
        stack[++top] = value;
}

void pop()
{
    if(top == -1)
        printf("Stack Underflow\n");
    else
        printf("Popped element: %d\n", stack[top--]);
}

int main()
{
    push(10);
    push(20);
    push(30);

    printf("Stack elements: ");
    for(int i = top; i >= 0; i--)
        printf("%d ", stack[i]);

    printf("\n");
    pop();

    return 0;
}




10. Perform enqueue and dequeue operations in a queue


#include <stdio.h>
#define MAX 5

int queue[MAX], front = 0, rear = -1;

void enqueue(int value)
{
    if(rear == MAX - 1)
        printf("Queue Overflow\n");
    else
        queue[++rear] = value;
}

void dequeue()
{
    if(front > rear)
        printf("Queue Underflow\n");
    else
        printf("Deleted element: %d\n", queue[front++]);
}

int main()
{
    enqueue(10);
    enqueue(20);
    enqueue(30);

    printf("Queue elements: ");
    for(int i = front; i <= rear; i++)
        printf("%d ", queue[i]);

    printf("\n");
    dequeue();

    return 0;
}



11. Create a binary search tree and display its elements

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *left, *right;
};

struct Node *insert(struct Node *root, int value)
{
    if(root == NULL)
    {
        root = malloc(sizeof(struct Node));
        if(root == NULL)
            exit(1);

        root->data = value;
        root->left = root->right = NULL;
    }
    else if(value < root->data)
        root->left = insert(root->left, value);
    else if(value > root->data)
        root->right = insert(root->right, value);

    return root;
}

void inorder(struct Node *root)
{
    if(root != NULL)
    {
        inorder(root->left);
        printf("%d ", root->data);
        inorder(root->right);
    }
}

void freeTree(struct Node *root)
{
    if(root != NULL)
    {
        freeTree(root->left);
        freeTree(root->right);
        free(root);
    }
}

int main()
{
    struct Node *root = NULL;
    int n, i, value;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("Enter element: ");
        scanf("%d", &value);
        root = insert(root, value);
    }

    printf("BST elements: ");
    inorder(root);

    freeTree(root);
    return 0;
}




12. Search for an element in an array

#include <stdio.h>

int main()
{
    int a[100], n, key, i, found = 0;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter element to search: ");
    scanf("%d", &key);

    for(i = 0; i < n; i++)
    {
        if(a[i] == key)
        {
            printf("Element found at position %d\n", i + 1);
            found = 1;
            break;
        }
    }

    if(!found)
        printf("Element not found\n");

    return 0;
}
